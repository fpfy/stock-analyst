### ⚙️ 高级配置

#### 1. 自定义限流策略
```python
class CustomAPILimiter(APILimiter):
    """自定义API限流器"""
    
    def __init__(self, config):
        super().__init__(
            max_calls_per_minute=config.get('minute_limit', 30),
            max_calls_per_hour=config.get('hour_limit', 200),
            max_calls_per_day=config.get('day_limit', 1000)
        )
        self.config = config
    
    def check_limit_with_priority(self, priority='normal'):
        """带优先级的限流检查"""
        if priority == 'high':
            # 高优先级任务放宽限制
            return self.minute_calls < self.max_calls_per_minute * 1.5
        elif priority == 'low':
            # 低优先级任务严格限制
            return self.minute_calls < self.max_calls_per_minute * 0.5
        else:
            return self.check_limit()
```

#### 2. 智能重试策略
```python
def smart_retry_api_call(api_func, max_retries=3, backoff_factor=2.0):
    """智能重试API调用"""
    
    for attempt in range(max_retries):
        try:
            result = api_func()
            return result
        except Exception as e:
            if attempt == max_retries - 1:
                raise
            
            # 指数退避
            wait_time = backoff_factor ** (attempt + 1)
            logger.warning(f'API调用失败，{wait_time}秒后重试: {e}')
            time.sleep(wait_time)
```

### 📊 监控和报警

#### 1. 实时监控
```python
class APIMonitorSystem:
    """API监控系统"""
    
    def __init__(self):
        self.monitors = {}
    
    def add_monitor(self, name):
        self.monitors[name] = APIMonitor()
        return self.monitors[name]
    
    def get_system_stats(self):
        """获取系统整体统计"""
        stats = {}
        for name, monitor in self.monitors.items():
            stats[name] = monitor.get_stats()
        return stats
    
    def check_system_health(self):
        """检查系统健康状态"""
        for name, monitor in self.monitors.items():
            stats = monitor.get_stats()
            if stats['success_rate'] < 0.9:
                logger.warning(f'{name} API成功率过低: {stats["success_rate"]:.2%}')
            if stats['recent_errors'] > 10:
                logger.warning(f'{name} API错误过多: {stats["recent_errors"]}')
```
